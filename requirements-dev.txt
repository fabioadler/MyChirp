# Complete development setup for this CHIRP build (Linux).
#
# NOTE ON wxPython: it is intentionally NOT listed here. On Debian 13
# (Python 3.13) there is no pip wheel for wxPython; use the system
# package instead and expose it to the venv:
#
#   sudo apt install python3-wxgtk4.0
#   python3 -m venv --system-site-packages .venv
#   .venv/bin/pip install -r requirements-dev.txt
#   .venv/bin/chirp
#
# The `-e .` line installs CHIRP itself in editable mode (its
# install_requires dependencies are also listed below explicitly for
# clarity). See README.md for the full usage guide.

-e .

# --- runtime dependencies (mirror of setup.py install_requires) ---
pyserial
requests
yattag
suds
lark

# --- test toolchain (mirror of test-requirements.txt plus ours) ---
pytest==7.1.3
pytest-xdist
pytest-html
six
pep8
pyyaml
# The system user-site has pytest-asyncio 0.24, which requires
# pytest>=8.2 and crashes when loaded by our pinned pytest 7.1.3.
# Installing a compatible version inside the venv shadows the system
# one and removes both the pip resolver conflict and the crash.
pytest-asyncio==0.21.2

# --- style checks (used by tools/cpep8.py) ---
flake8

# --- locale catalog compilation ---
# gettext tools (msgfmt) are not available in this environment, so the
# pt_BR/en_US/zh_CN/ja_JP catalogs are compiled with polib (see the
# language section of README.md).
polib

# --- CH341A programmer support (read-only chip tool) ---
pyusb
